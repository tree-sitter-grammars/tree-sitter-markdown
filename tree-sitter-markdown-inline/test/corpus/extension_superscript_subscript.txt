================================================================================
Basic superscript parsing
================================================================================
^superscript^

-------------------------------------------------------------------------------
(inline
  (superscript
    (emphasis_delimiter)
    (emphasis_delimiter)))

================================================================================
Basic subscript parsing
================================================================================
~subscript~

-------------------------------------------------------------------------------
(inline
  (subscript
    (emphasis_delimiter)
    (emphasis_delimiter)))

================================================================================
Double tildes are parsed as strikethrough, not subscript
================================================================================
~~strike~~

-------------------------------------------------------------------------------
(inline
  (strikethrough
    (emphasis_delimiter)
    (strikethrough
      (emphasis_delimiter)
      (emphasis_delimiter))
    (emphasis_delimiter)))
